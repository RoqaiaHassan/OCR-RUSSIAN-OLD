import os
import requests

# ------------------ الإعدادات ------------------
ATTACHMENT_ID = "5dd3ea5274d5af0dd456868e"   # معرّف المخطوطة على الموقع
SAVE_DIR = "/content/sluzhebnik_manuscript"
BASE_URL = f"https://kp.rusneb.ru/attachments_new/attachment/{ATTACHMENT_ID}/preview-page-{{}}.jpg"
HEADERS = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) Chrome/124.0.0.0"}
MAX_PAGES = 130   # المخطوطة 120 ورقة، بنحط هامش أمان زيادة شوية
TARGET_TOTAL = 110  # التوقف بعد الوصول لـ 100 صورة بالظبط

os.makedirs(SAVE_DIR, exist_ok=True)

# ------------------ تحميل كل الصفحات ------------------
downloaded = 0
consecutive_failures = 0

for page_num in range(1, MAX_PAGES + 1):
    if downloaded >= TARGET_TOTAL:
        print(f"  تم الوصول للهدف ({TARGET_TOTAL} صورة)، إيقاف التحميل.")
        break
    url = BASE_URL.format(page_num)
    try:
        resp = requests.get(url, headers=HEADERS, timeout=20)
        content_type = resp.headers.get("Content-Type", "")

        if resp.status_code == 200 and "image" in content_type and len(resp.content) > 1000:
            filepath = os.path.join(SAVE_DIR, f"page_{page_num:03d}.jpg")
            with open(filepath, "wb") as f:
                f.write(resp.content)
            downloaded += 1
            consecutive_failures = 0
            if downloaded % 10 == 0:
                print(f"  تم تحميل {downloaded} صفحة...")
        else:
            print(f"  توقفت الصفحة {page_num}: status={resp.status_code} type={content_type}")
            consecutive_failures += 1
            # لو فشلت 5 صفحات متتالية، غالبًا وصلنا لنهاية المخطوطة
            if consecutive_failures >= 5:
                print("  يبدو أننا وصلنا لنهاية المخطوطة، إيقاف التحميل.")
                break
    except Exception as e:
        print(f"  خطأ في الصفحة {page_num}: {e}")
        consecutive_failures += 1
        if consecutive_failures >= 5:
            break

print(f"\n✅ تم تحميل {downloaded} صفحة إلى: {SAVE_DIR}")

# ------------------ ضغط المجلد ------------------
import shutil
shutil.make_archive(SAVE_DIR, "zip", SAVE_DIR)
print(f"تم إنشاء ملف مضغوط: {SAVE_DIR}.zip")
