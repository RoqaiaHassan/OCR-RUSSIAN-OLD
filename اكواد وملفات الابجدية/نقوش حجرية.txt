# 6. دالة لتنزيل الصور من صفحة معينة
def download_images_from_page(page_url, save_folder):
    try:
        # محاكاة متصفح حقيقي
        headers = {'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'}
        response = requests.get(page_url, headers=headers, timeout=15)
        response.raise_for_status()  # التأكد من نجاح التحميل

        soup = BeautifulSoup(response.text, 'html.parser')
        img_tags = soup.find_all('img')

        downloaded = 0
        for img in img_tags:
            img_url = img.get('src')
            if not img_url:
                continue

            # تحويل الرابط النسبي إلى رابط كامل
            img_url = urljoin(page_url, img_url)

            # تجاهل الصور الصغيرة جدًا (أيقونات، شعارات)
            if 'avatar' in img_url or 'icon' in img_url or 'logo' in img_url:
                continue

            # استخراج اسم الملف
            img_name = os.path.basename(urlparse(img_url).path)
            if not img_name or len(img_name) < 5:
                img_name = f"image_{random.randint(1000,9999)}.jpg"

            # التأكد من الامتداد
            if not img_name.lower().endswith(('.png', '.jpg', '.jpeg', '.gif', '.webp')):
                img_name += '.jpg'

            # حفظ الصورة
            img_path = os.path.join(save_folder, img_name)

            # تجنب تكرار التحميل
            if os.path.exists(img_path):
                continue

            try:
                img_data = requests.get(img_url, headers=headers, timeout=10)
                if img_data.status_code == 200:
                    with open(img_path, 'wb') as f:
                        f.write(img_data.content)
                    downloaded += 1
                    print(f"  ✅ تم تحميل: {img_name}")
                else:
                    print(f"  ⚠️ فشل تحميل: {img_url} (كود {img_data.status_code})")
            except Exception as e:
                print(f"  ⚠️ خطأ في تحميل {img_url}: {e}")

            # تأخير بسيط بين الصور
            time.sleep(random.uniform(0.5, 1.5))

        return downloaded

    except Exception as e:
        print(f"❌ خطأ في معالجة الصفحة {page_url}: {e}")
        return 0

# 7. تنفيذ السكربت على جميع الصفحات
total_images = 0
for i, url in enumerate(urls_to_scrape):
    # إنشاء مجلد فرعي لكل صفحة
    folder_name = f"page_{i+1}"
    save_folder = os.path.join(base_dir, folder_name)
    os.makedirs(save_folder, exist_ok=True)

    print(f"\n📄 [{i+1}/{len(urls_to_scrape)}] معالجة: {url}")
    print(f"📂 الحفظ في: {folder_name}")

    # تنزيل الصور من الصفحة
    count = download_images_from_page(url, save_folder)
    total_images += count
    print(f"✅ تم تحميل {count} صورة من هذه الصفحة")

    # تأخير بين الصفحات لتجنب الحظر
    if i < len(urls_to_scrape) - 1:
        delay = random.uniform(5, 10)
        print(f"⏳ انتظار {delay:.1f} ثواني قبل الصفحة التالية...")
        time.sleep(delay)
